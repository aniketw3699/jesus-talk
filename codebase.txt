def parse_and_sanitize(raw_text: str, current_psyche: str) -> tuple[str, str]:
    cleaned = raw_text
    if "</think>" in cleaned:
        cleaned = cleaned.split("</think>")[-1].strip()
    cleaned = re.sub(r'<think>[\s\S]*?<\/think>', '', cleaned, flags=re.IGNORECASE)
    cleaned = re.sub(r'<think>[\s\S]*', '', cleaned, flags=re.IGNORECASE)

    updated_psyche = current_psyche
    reply_body = cleaned

    psyche_match = re.search(r'PSYCHE:\s*(.*?)(?=\n|REPLY:|$)', cleaned, flags=re.IGNORECASE)
    if psyche_match:
        extracted = psyche_match.group(1).strip()
        if len(extracted) > 3:
            updated_psyche = extracted

    if "REPLY:" in cleaned:
        reply_body = cleaned.split("REPLY:")[-1].strip()
    else:
        reply_body = re.sub(r'PSYCHE:\s*.*?\n', '', cleaned, flags=re.IGNORECASE).strip()

    reply_body = re.sub(r'^(Draft|Response):\s*', '', reply_body, flags=re.IGNORECASE)
    reply_body = re.sub(r'^\d+\.\s*\*\*[\s\S]*?\n\n', '', reply_body)
    reply_body = re.sub(r'\s*—\s*', ', ', reply_body)
    reply_body = re.sub(r'\s*–\s*', ', ', reply_body)
    reply_body = re.sub(r',\s*,', ',', reply_body)

    return reply_body.strip(), updated_psyche.strip()

async def execute_chat(payload: dict) -> dict:
    if not GROQ_API_KEY or not client:
        raise HTTPException(status_code=500, detail="GROQ_API_KEY is not configured.")

    user_msg = str(payload.get("message", "")).strip()
    user_name = str(payload.get("userName", "beloved"))
    user_psyche = str(payload.get("userPsyche", "A soul seeking guidance"))
    history = payload.get("history", [])

    if not user_msg:
        return {"reply": "Speak your heart, I am listening.", "updatedPsyche": user_psyche}

    if re.match(r'^(\d+\s*[\+\-\*\/\^xX]\s*\d+|\bcalculate\b|\bwhat is \d+)', user_msg.lower()):
        return {
            "reply": "Human numbers measure the dust of the earth, yet they cannot measure the boundless grace given to you. Look beyond calculations, and let us attend to your soul.",
            "updatedPsyche": "Pondering calculations over spiritual matters"
        }

    system_prompt = SYSTEM_PROMPT_TEMPLATE.format(user_name=user_name, user_psyche=user_psyche)
    messages = [{"role": "system", "content": system_prompt}]

    for item in history[-6:]:
        content = item.get("content", "").strip() if isinstance(item, dict) else ""
        role = item.get("role", "user") if isinstance(item, dict) else "user"
        if content:
            mapped_role = "assistant" if role in ["assistant", "bot"] else "user"
            messages.append({"role": mapped_role, "content": content})

    messages.append({"role": "user", "content": user_msg})

    models_to_try = [
        "openai/gpt-oss-120b",
        "openai/gpt-oss-20b",
        "groq/compound",
        "qwen/qwen3.6-27b"
    ]

    raw_output = ""
    for model_name in models_to_try:
        try:
            res = client.chat.completions.create(
                model=model_name,
                messages=messages,
                temperature=0.45,
                max_tokens=550
            )
            content = res.choices[0].message.content or ""
            if content.strip():
                raw_output = content
                break
        except Exception:
            continue

    if not raw_output:
        raw_output = "REPLY: Peace be with you. Lay down whatever burdens your heart, for I am listening."

    final_reply, evolved_psyche = parse_and_sanitize(raw_output, user_psyche)
    return {
        "reply": final_reply,
        "updatedPsyche": evolved_psyche
    }

def update_sitemap(published_list):
    today_ist = datetime.now(IST).strftime('%Y-%m-%d')
    urls = [
        f"""  <url>
    <loc>{DOMAINS_URL}/</loc>
    <lastmod>{today_ist}</lastmod>
    <changefreq>daily</changefreq>
    <priority>1.0</priority>
  </url>""",
        f"""  <url>
    <loc>{DOMAINS_URL}/blogs.html</loc>
    <lastmod>{today_ist}</lastmod>
    <changefreq>daily</changefreq>
    <priority>0.9</priority>
  </url>"""
    ]
    for item in published_list:
        urls.append(f"""  <url>
    <loc>{DOMAINS_URL}/prayers/{item['slug']}.html</loc>
    <lastmod>{today_ist}</lastmod>
    <changefreq>monthly</changefreq>
    <priority>0.8</priority>
  </url>""")

    sitemap_content = f"""<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
{chr(10).join(urls)}
</urlset>
"""
    with open(SITEMAP_FILE, "w", encoding="utf-8") as f:
        f.write(sitemap_content)
